/**

*     [16, 16, 1]. // Instance Bounding Box

*     [0, 0, 0], // Instance offset in the set

*/
STEPS = [

/*
* Main Module Call
*/
mb_block__com__martianmicro__anyclosure__DefaultEnclosure(
    config = mb_config,
    settings = [
        ["baseColor", baseColor],
        ["highlightColor", highlightColor],
        ["mode", mode],
        ["step", step],
        ["instanceId", instanceId],
        ["combined", true],
        ["assembly", assembly]
    ]
);

/*
* Main Module Definition
*/
module mb_block__com__martianmicro__anyclosure__DefaultEnclosure(config = undef, settings = undef){
    mode = mb_param(config, settings, "mode", default="total");

    if(mode == "total"){
        mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Total(config = config, settings = settings);
    }
    else if(mode == "print"){
        mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Print(config = config, settings = settings);
    }
    else if(mode == "step"){
        mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Step(config = config, settings = settings);
    }
    else if(mode == "instance"){
        mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Instance(config = config, settings = settings);
    }
}

/*
* Sub Module Definition: Assembled Enclosure
*/
module mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Total(config = undef, settings = undef){
    blockId = mb_param_id(config, settings, "mm.anyclosure.default_enclosure.total");
    baseColor = mb_param_baseColor(config, settings);
    align = mb_param_align(config, settings);
    direction = mb_param_direction(config, settings);
    offset = mb_param_offset(config, settings);
    
    mb_block(
        config = config,
        settings = [
            ["id", blockId],
            ["base", false],
            ["studs", false],
            ["align", align],
            ["offset", offset],
            ["direction", direction],
            ["size",  mb_parts_total_size(STEPS)] // <- Border Box must be calculated
        ]
    ){
        for(i = [0 : len(STEPS) - 1]){
            blockSettings = [
                ["id", mb_block_id(blockId, STEPS[i][0])],
                ["instanceId", STEPS[i][0]],
                
                // Set Options
                ["size", STEPS[i][1]],
                ["direction", STEPS[i][2]],
                ["offset", STEPS[i][3]],

                // View Options
                ["baseColor", baseColor],
                ["assembly", "merged"]
            ];
            mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Instance(config = config, settings = blockSettings);
        }
    }
}

/*
* Sub Module Definition: 3D Print View with single parts
*/
module mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Print(config = undef, settings = undef){
    blockId = mb_param_id(config, settings, "mm.anyclosure.default_enclosure.print");
    baseColor = mb_param_baseColor(config, settings);
    align = mb_param_align(config, settings);
    direction = mb_param_direction(config, settings);
    offset = mb_param_offset(config, settings);
    assembly = mb_param_assembly(config, settings);

    mb_block(
        config = config,
        settings = [
            ["id", blockId],
            ["base", false],
            ["studs", false],
            ["align", align],
            ["offset", offset],
            ["direction", direction],
            ["size",  mb_parts_total_size(STEPS)] // <- Border Box must be calculated
        ]
    ){
        for(i = [0 : len(STEPS) - 1]){
            blockSettings = [
                ["id", mb_block_id(blockId, STEPS[i][0])],
                ["instanceId", STEPS[i][0]],

                ["size", STEPS[i][1]],

                // View Options
                ["offset", mb_set_step_print_position(assembly, STEPS, i)],
                ["assembly", assembly],
                ["baseColor", baseColor]
            ];
            mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Instance(config = config, settings = blockSettings);
        }
    }
}

/*
* Sub Module Definition: Step by Step View
*/
module mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Step(config = undef, settings = undef){
    blockId = mb_param_id(config, settings, "mm.anyclosure.default_enclosure.step");
    step = mb_param(config, settings, "step");
    combined = mb_param(config, settings, "combined", false);
    highlightColor = mb_param(config, settings, "highlightColor", "#EAC645");

    baseColor = mb_param_baseColor(config, settings);

    align = mb_param_align(config, settings);
    direction = mb_param_direction(config, settings);
    offset = mb_param_offset(config, settings);
    
    
    instanceId = STEPS[step][0];

    if(instanceId != undef){
        mb_block(
            config = config,
            settings = [
                ["id", blockId],
                ["base", false],
                ["studs", false],
                ["align", align],
                ["offset", offset],
                ["direction", direction],
                ["size",  mb_parts_total_size(STEPS)] // <- Border Box must be calculated
            ]
        ){
            if(combined){
                for(i = [0 : step - 1]){
                    blockSettings = [
                        ["id", mb_block_id(blockId, STEPS[i][0])],
                        ["instanceId", STEPS[i][0]],
                        
                        // Set Options
                        ["size", STEPS[i][1]],
                        ["direction", STEPS[i][2]],
                        ["offset", STEPS[i][3]],

                        // View Options
                        ["baseColor", baseColor],
                        ["assembly", "merged"]
                    ];
                    mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Instance(config = config, settings = blockSettings);
                }
            }

            //Instance
            mb_block__com__martianmicro__anyclosure__DefaultEnclosure__Instance(
                config = config,
                settings = [
                    ["id", mb_block_id(blockId, instanceId)],
                    ["instanceId", instanceId],
                    
                    // Set Options
                    ["size", STEPS[step][1]],
                    ["direction", STEPS[step][2]],
                    ["offset", STEPS[step][3]],

                    // View Options
                    ["baseColor", combined ? highlightColor : baseColor],
                    ["assembly", "merged"]
                ]
            );
        }
    }
}
