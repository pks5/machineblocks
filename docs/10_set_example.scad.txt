/**
* List of instances used in this set, sorted by instruction step index. 
* 
* Array of Array[5]
*
* Example (STEPS[0]):
*
* [
*     "00001", // Instance ID, can be any string
*     [16, 16, 1]. // Instance Bounding Box
*     false, // Whether Instance supports assembly
*     "west", // Instance direction in the set
*     [0, 0, 0], // Instance offset in the set
* ]
*/
STEPS = [
    ["00001", [16, 16, 1], false, "west", [0, 0, 0]],

];

/*
* Main Module Call
*/
mb_block__mm__anyclosure__default_enclosure(
    config = mb_config,
    settings = [
        ["baseColor", baseColor],
        ["highlightColor", highlightColor],
        ["mode", mode],
        ["step", step],
        ["instanceId", instanceId],
        ["combined", true],
        ["assembly", assembly]
    ]
);

/*
* Main Module Definition
*/
module mb_block__mm__anyclosure__default_enclosure(config = undef, settings = undef){
    mode = mb_param(config, settings, "mode", default="total");

    if(mode == "total"){
        mb_block__mm__anyclosure__default_enclosure__total(config = config, settings = settings);
    }
    else if(mode == "print"){
        mb_block__mm__anyclosure__default_enclosure__print(config = config, settings = settings);
    }
    else if(mode == "step"){
        mb_block__mm__anyclosure__default_enclosure__step(config = config, settings = settings);
    }
    else if(mode == "instance"){
        mb_block__mm__anyclosure__default_enclosure__instance(config = config, settings = settings);
    }
}

/*
* Sub Module Definition: Assembled Enclosure
*/
module mb_block__mm__anyclosure__default_enclosure__total(config = undef, settings = undef){
    baseColor = mb_param_baseColor(config, settings);
    align = mb_param_align(config, settings);
    direction = mb_param_direction(config, settings);
    offset = mb_param_offset(config, settings);
    

    mb_block(
        config = config,
        settings = [
            ["base", false],
            ["studs", false],
            ["align", align],
            ["offset", offset],
            ["direction", direction],
            ["size", [16,16,9]] // <- Border Box must be calculated
        ]
    ){
        for(i = [0 : len(STEPS) - 1]){
            blockSettings = [
                ["step", i],

                // View Options
                ["baseColor", baseColor],
                ["assembly", "merged"]
            ];
            mb_block__mm__anyclosure__default_enclosure__step(config = config, settings = blockSettings);
        }
    }
}

/*
* Sub Module Definition: 3D Print View with single parts
*/
module mb_block__mm__anyclosure__default_enclosure__print(config = undef, settings = undef){
    baseColor = mb_param_baseColor(config, settings);
    align = mb_param_align(config, settings);
    direction = mb_param_direction(config, settings);
    offset = mb_param_offset(config, settings);
    assembly = mb_param_assembly(config, settings);

    mb_block(
        config = config,
        settings = [
            ["base", false],
            ["studs", false],
            ["align", align],
            ["offset", offset],
            ["direction", direction],
            ["size", [16,16,9]] // <- Border Box must be calculated
        ]
    ){
        for(i = [0 : len(STEPS) - 1]){
            blockSettings = [
                ["instanceId", STEPS[i][0]],

                // View Options
                ["offset", mb_set_step_print_position(assembly, STEPS, i)],
                ["assembly", assembly],
                ["baseColor", baseColor]
            ];
            mb_block__mm__anyclosure__default_enclosure__instance(config = config, settings = blockSettings);
        }
    }
}

/*
* Sub Module Definition: Step by Step View
*/
module mb_block__mm__anyclosure__default_enclosure__step(config = undef, settings = undef){
    step = mb_param(config, settings, "step");
    combined = mb_param(config, settings, "combined", false);
    highlightColor = mb_param(config, settings, "highlightColor", "#EAC645");

    baseColor = mb_param_baseColor(config, settings);

    align = mb_param_align(config, settings);
    direction = mb_param_direction(config, settings);
    offset = mb_param_offset(config, settings);
    
    
    instanceId = STEPS[step][0];

    if(instanceId != undef){
        mb_block(
            config = config,
            settings = [
                ["base", false],
                ["studs", false],
                ["align", align],
                ["offset", offset],
                ["direction", direction],
                ["size", [16,16,9]] // <- Border Box must be calculated
            ]
        ){
            if(combined){
                for(i = [0 : step - 1]){
                    blockSettings = [
                        ["step", i],

                        // View Options
                        ["baseColor", baseColor],
                        ["assembly", "merged"]
                    ];
                    mb_block__mm__anyclosure__default_enclosure__step(config = config, settings = blockSettings);
                }
            }

            //Instance
            mb_block__mm__anyclosure__default_enclosure__instance(
                config = config,
                settings = [
                    ["instanceId", instanceId],
                    
                    // Set Options
                    ["direction", STEPS[step][3]],
                    ["offset", STEPS[step][4]],

                    // View Options
                    ["baseColor", combined ? highlightColor : baseColor],
                    ["assembly", "merged"]
                ]
            );
        }
    }
}
